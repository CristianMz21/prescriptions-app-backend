sequenceDiagram
    participant C Client
    participant AC AuthController
    participant AS AuthService
    participant PR PrismaService
    participant JWT JwtService
    participant DB PostgreSQL
    C->>AC POST /auth/login {email, password}
    AC->>AS login(dto)
    AS->>PR user.findUnique(email)
    PR->>DB SELECT WHERE email=?
    DB-->>PR User
    alt usuario no existe
        AS-->>C 401 "Credenciales invalidas"
    end
    AS->>AS bcrypt.compare
    alt contrasena incorrecta
        AS-->>C 401
    end
    AS->>JWT sign(payload)
    JWT-->>AS accessToken
    AS->>AS crypto.randomBytes(64)
    AS->>PR refreshToken.create
    PR->>DB INSERT
    AS-->>C {accessToken, refreshToken, user}

sequenceDiagram
    participant D Doctor
    participant PC PrescriptionsController
    participant PS PrescriptionsService
    participant PR PrismaService
    participant DB PostgreSQL
    D->>PC POST /prescriptions {patientId, items}
    PC->>PS create(dto, authorId)
    alt patientId provided
        PS->>PR patient.findUnique(patientId)
    else patientEmail provided
        PS->>PR patient.findFirst(user.email=patientEmail)
    end
    alt paciente no encontrado
        PS-->>D 404
    end
    PS->>PR prescription.create
    PR->>DB INSERT prescription + items
    PS-->>D Prescription 201
    participant Pt Patient
    participant PtC PrescriptionsController
    participant PtS PrescriptionsService
    Pt->>PtC GET /prescriptions
    PtC->>PtS findAll(patientId)
    PtS->>PR prescription.findMany(where patientId)
    PR->>DB SELECT
    PtS-->>Pt {data, meta}
    Pt->>PtC PATCH /prescriptions/:id/consume
    PtC->>PtS consume(id, patientId)
    alt no es el dueno
        PtS-->>Pt 403
    end
    PtS->>PR prescription.update status=consumed
    PtS-->>Pt Prescription actualizada
